def calculate_total_cost(prices):
    total = 0
    for price in prices:
        if isinstance(price, (int, float)):
            total += price
        elif isinstance(price, str):
            try:
                total += float(price)
            except ValueError:
                pass
    return total

# Example usage:
prices = [10.5, '20', 30, 'abc']
print(calculate_total_cost(prices))  # Output: 60.5
